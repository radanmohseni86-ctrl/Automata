\chapter{ساده‌سازی گرامرها و فرم‌های نرمال}

گرامرهای مستقل از متن می‌توانند شکل‌های بسیار متنوعی داشته باشند. برای اثبات قضیه‌ها و ساختن الگوریتم‌ها (مثلاً الگوریتم \lr{CYK} در فصل بعد) خیلی بهتر است گرامر شکل ساده و استانداردی داشته باشد. در این فصل نشان می‌دهیم سه نوع قاعده را همیشه می‌توان حذف کرد: قاعده‌های \textbf{بی‌فایده}، \textbf{قاعده‌های $\eps$} و \textbf{قاعده‌های واحد}. سپس دو \textbf{فرم نرمال} معروف را معرفی می‌کنیم: فرم نرمال چامسکی و فرم نرمال گرایباخ.

\begin{note}
«ساده‌سازی» لزوماً به معنای کم شدن تعداد قاعده‌ها نیست؛ گاهی تعداد قاعده‌ها زیاد هم می‌شود. هدف این است که قاعده‌ها \textbf{شکل} ساده‌تری داشته باشند.
\end{note}

در این فصل گرامر را با نمادگذاری $G=(V,T,S,P)$ می‌نویسیم ($T$ پایانه‌ها و $P$ قاعده‌ها).

\section{یک قاعده جایگزینی مفید}

\begin{theorem}\label{thm:subst}
فرض کنید گرامر $G=(V,T,S,P)$ قاعده‌های زیر را داشته باشد:
\[(\ast)\ \ A\to x_1Bx_2,\qquad(\ast\ast)\ \ B\to y_1\mid y_2\mid\cdots\mid y_n,\]
که $A\neq B$ و $(\ast\ast)$ همه قاعده‌هایی است که $B$ در سمت چپشان است. اگر $\widehat G=(V,T,S,\widehat P)$ را با حذف قاعده $(\ast)$ و اضافه کردن قاعده‌های
\[A\to x_1y_1x_2\mid x_1y_2x_2\mid\cdots\mid x_1y_nx_2\]
بسازیم، آنگاه $L(\widehat G)=L(G)$.
\end{theorem}
\begin{proof}
\textbf{$L(\widehat G)\subseteq L(G)$:} هر قاعده جدید $A\to x_1y_ix_2$ را می‌توان در $G$ با دو گام شبیه‌سازی کرد: $A\Rightarrow x_1Bx_2\Rightarrow x_1y_ix_2$. پس هر اشتقاق در $\widehat G$ را می‌توان به یک اشتقاق در $G$ تبدیل کرد.

\textbf{$L(G)\subseteq L(\widehat G)$:} اشتقاقی از $w$ در $G$ را در نظر بگیرید. اگر از $(\ast)$ استفاده نکند، در $\widehat G$ هم معتبر است. اگر استفاده کند، متغیر $B$ که با آن ساخته شده بالاخره باید با یکی از $y_i$ها جایگزین شود (چون $w$ متغیری ندارد و $(\ast\ast)$ تنها راه جایگزینی $B$ است). ترتیب گام‌ها در گرامر مستقل از متن مهم نیست (جایگزینی‌های مستقل را می‌توان جابه‌جا کرد)، پس می‌توانیم جایگزینی این $B$ را درست بعد از به کار بردن $(\ast)$ انجام دهیم. این دو گام با یک گام $A\to x_1y_ix_2$ در $\widehat G$ برابرند. با تکرار این کار برای هر استفاده از $(\ast)$، اشتقاقی برای $w$ در $\widehat G$ به دست می‌آید.
\end{proof}

\section{حذف متغیرها و قاعده‌های بی‌فایده}

\begin{definition}[متغیر مفید و بی‌فایده]
متغیر $A$ در گرامر $G=(V,T,S,P)$ \textbf{مفید} است اگر دست‌کم یک $w\in L(G)$ وجود داشته باشد که
\[S\der xAy\der w\qquad (x,y\in(V\cup T)^*).\]
یعنی $A$ در اشتقاق دست‌کم یک رشته پایانه‌ای شرکت کند. متغیری که مفید نیست \textbf{بی‌فایده} است و قاعده‌ای که یک متغیر بی‌فایده دارد \textbf{قاعده بی‌فایده} است.
\end{definition}

یک متغیر به دو دلیل ممکن است بی‌فایده باشد:
\begin{enumerate}
\item از متغیر شروع \textbf{قابل دسترسی نیست} (هیچ اشتقاقی از $S$ به آن نمی‌رسد)؛
\item \textbf{هیچ رشته پایانه‌ای تولید نمی‌کند}.
\end{enumerate}

\begin{theorem}
برای هر گرامر مستقل از متن $G$ (با $L(G)\neq\emptyset$) گرامر هم‌ارزی $\widehat G$ وجود دارد که هیچ متغیر یا قاعده بی‌فایده‌ای ندارد.
\end{theorem}

\begin{proof}
حذف در دو مرحله و \textbf{به همین ترتیب} انجام می‌شود.

\textbf{مرحله ۱: حذف متغیرهایی که رشته پایانه‌ای تولید نمی‌کنند.} مجموعه $V_1$ از متغیرهای «مولد» را این‌طور می‌سازیم: ابتدا هر $A$ که قاعده‌ای مثل $A\to x$ با $x\in T^*$ دارد را در $V_1$ بگذار. سپس تا وقتی تغییری رخ می‌دهد، هر $A$ که قاعده‌ای مثل $A\to x_1x_2\cdots x_k$ دارد که هر $x_i$ یا پایانه است یا در $V_1$، را به $V_1$ اضافه کن. این حلقه حداکثر $\abs{V}$ بار تکرار می‌شود. با استقرا روی طول اشتقاق روشن است که $A\in V_1$ اگر و تنها اگر $A$ یک رشته پایانه‌ای تولید کند. همه متغیرهای خارج از $V_1$ و همه قاعده‌هایی که آن‌ها را دارند حذف می‌کنیم. زبان تغییر نمی‌کند، چون هیچ اشتقاق کاملی از آن متغیرها استفاده نمی‌کند (اگر استفاده می‌کرد، آن متغیر یک رشته پایانه‌ای تولید می‌کرد).

\textbf{مرحله ۲: حذف متغیرهای غیرقابل دسترس.} \textbf{گراف وابستگی} می‌سازیم: رأس‌ها متغیرها هستند و از $A$ به $B$ یال هست اگر قاعده‌ای مثل $A\to xBy$ وجود داشته باشد. هر متغیری که در این گراف از $S$ مسیری به آن نیست، همراه با قاعده‌هایش حذف می‌شود. باز هم زبان تغییر نمی‌کند.

بعد از این دو مرحله هر متغیر باقی‌مانده $A$ از $S$ قابل دسترسی است ($S\der xAy$) و هر متغیر (از جمله $A$ و متغیرهای درون $x$ و $y$) یک رشته پایانه‌ای تولید می‌کند. پس $S\der xAy\der w$ برای یک $w\in T^*$، یعنی $A$ مفید است.
\end{proof}

\begin{warning}
ترتیب مراحل مهم است. اگر اول متغیرهای غیرقابل دسترس را حذف کنیم، ممکن است حذف متغیرهای نامولد متغیرهای جدیدی را غیرقابل دسترس کند.
\end{warning}

\begin{example}
\[S\to aS\mid A\mid C,\qquad A\to a,\qquad B\to aa,\qquad C\to aCb.\]
\textbf{مرحله ۱:} $C$ هیچ رشته پایانه‌ای تولید نمی‌کند (هر بار $C$ دوباره ظاهر می‌شود). پس $C$ و قاعده‌های $S\to C$ و $C\to aCb$ حذف می‌شوند:
\[S\to aS\mid A,\qquad A\to a,\qquad B\to aa.\]
\textbf{مرحله ۲:} در گراف وابستگی یال‌های $S\to S$ و $S\to A$ را داریم و هیچ یالی به $B$ نمی‌رسد. پس $B$ حذف می‌شود:
\[S\to aS\mid A,\qquad A\to a.\]
\end{example}

\section{حذف قاعده‌های $\eps$}

\begin{definition}[قاعده $\eps$ و متغیر تهی‌پذیر]
قاعده‌ای به شکل $A\to\eps$ یک \textbf{قاعده $\eps$} است. متغیر $A$ که برای آن $A\der\eps$ ممکن باشد \textbf{تهی‌پذیر} (\lr{nullable}) نام دارد.
\end{definition}

\begin{theorem}
برای هر گرامر مستقل از متن $G$ که $\eps\notin L(G)$، گرامر هم‌ارزی $\widehat G$ بدون قاعده $\eps$ وجود دارد.
\end{theorem}

\begin{proof}
\textbf{گام ۱: یافتن متغیرهای تهی‌پذیر.} مجموعه $V_N$ را این‌طور می‌سازیم:
\begin{enumerate}
\item برای هر قاعده $A\to\eps$، $A$ را در $V_N$ بگذار.
\item تا وقتی تغییری رخ می‌دهد: برای هر قاعده $B\to A_1A_2\cdots A_n$ که همه $A_1,\dots,A_n$ در $V_N$ باشند، $B$ را به $V_N$ اضافه کن.
\end{enumerate}

\textbf{گام ۲: ساختن قاعده‌های جدید.} هر قاعده به شکل $A\to x_1x_2\cdots x_m$ با $m\ge1$ (که $x_i\in V\cup T$) را در $\widehat P$ می‌گذاریم و همراه آن، \textbf{همه نسخه‌هایی} را که از حذف هر ترکیبی از متغیرهای تهی‌پذیر $x_i$ به دست می‌آیند. تنها استثنا: اگر همه $x_i$ها تهی‌پذیر باشند، نسخه‌ای که همه حذف شده‌اند (یعنی $A\to\eps$) را اضافه نمی‌کنیم. قاعده‌های $\eps$ اصلی هم کنار گذاشته می‌شوند.

\textbf{درستی.} هر قاعده جدید در $G$ قابل شبیه‌سازی است: متغیرهای حذف‌شده تهی‌پذیرند و در $G$ می‌توان آن‌ها را به $\eps$ تبدیل کرد؛ پس $L(\widehat G)\subseteq L(G)$. برعکس، یک درخت تجزیه برای $w\in L(G)$ (که $w\neq\eps$) را در نظر بگیرید. هر زیردرختی که فقط $\eps$ تولید می‌کند را حذف کنید و در گره پدر آن، به جای قاعده اصلی از نسخه‌ای استفاده کنید که آن متغیر حذف شده است. این نسخه در $\widehat P$ هست، چون آن متغیر تهی‌پذیر است و چون آن گره پدر چیزی ناتهی تولید می‌کند، همه فرزندانش حذف نشده‌اند. نتیجه یک درخت تجزیه معتبر در $\widehat G$ برای $w$ است. پس $L(G)\subseteq L(\widehat G)$.
\end{proof}

\begin{example}
\[S\to ABaC,\quad A\to BC,\quad B\to b\mid\eps,\quad C\to D\mid\eps,\quad D\to d.\]
متغیرهای تهی‌پذیر: ابتدا $B$ و $C$ (قاعده $\eps$ دارند)، سپس $A$ (چون $A\to BC$ و هر دو تهی‌پذیرند). پس $V_N=\{A,B,C\}$. گرامر جدید:
\[\begin{array}{l}
S\to ABaC\mid BaC\mid AaC\mid ABa\mid aC\mid Aa\mid Ba\mid a,\\
A\to B\mid C\mid BC,\qquad B\to b,\qquad C\to D,\qquad D\to d.
\end{array}\]
\end{example}

\section{حذف قاعده‌های واحد}

\begin{definition}[قاعده واحد]
قاعده‌ای به شکل $A\to B$ که $A,B\in V$ یک \textbf{قاعده واحد} است.
\end{definition}

\begin{theorem}
برای هر گرامر مستقل از متن $G$ بدون قاعده $\eps$، گرامر هم‌ارزی $\widehat G$ بدون قاعده واحد وجود دارد.
\end{theorem}

\begin{proof}
\begin{enumerate}
\item قاعده‌های به شکل $A\to A$ را حذف کنید؛ آن‌ها هیچ اثری ندارند.
\item برای هر متغیر $A$ همه متغیرهای $B$ را پیدا کنید که $A\der B$ فقط با قاعده‌های واحد. برای این کار گراف وابستگی می‌سازیم که رأس‌هایش متغیرها هستند و برای هر قاعده واحد $C\to D$ یالی از $C$ به $D$ دارد. $A\der B$ اگر و تنها اگر مسیری از $A$ به $B$ باشد.
\item همه قاعده‌های غیرواحد $P$ را در $\widehat P$ بگذارید.
\item برای هر زوج $A\der B$، اگر $B\to y_1\mid\cdots\mid y_n$ همه قاعده‌های غیرواحد $B$ در $\widehat P$ باشند، قاعده‌های $A\to y_1\mid\cdots\mid y_n$ را به $\widehat P$ اضافه کنید.
\end{enumerate}
\textbf{درستی.} هر قاعده جدید $A\to y_i$ در $G$ با $A\der B\Rightarrow y_i$ قابل شبیه‌سازی است. برعکس، در هر اشتقاق چپ‌ترین در $G$، هر دنباله از قاعده‌های واحد پشت سر هم مثل $A\Rightarrow B_1\Rightarrow\cdots\Rightarrow B_k\Rightarrow y$ (که آخری غیرواحد است) با یک گام $A\to y$ در $\widehat G$ جایگزین می‌شود. پس دو گرامر یک زبان را تولید می‌کنند.
\end{proof}

\begin{example}
\[S\to Aa\mid B,\qquad B\to A\mid bb,\qquad A\to a\mid bc\mid B.\]
گراف وابستگی قاعده‌های واحد یال‌های $S\to B$، $B\to A$ و $A\to B$ را دارد. پس $S\der A$، $S\der B$، $A\der B$ و $B\der A$. قاعده‌های غیرواحد:
\[S\to Aa,\qquad B\to bb,\qquad A\to a\mid bc.\]
حالا قاعده‌های جدید را اضافه می‌کنیم:
\begin{itemize}
\item از $S\der A$: \ $S\to a\mid bc$،
\item از $S\der B$: \ $S\to bb$،
\item از $A\der B$: \ $A\to bb$،
\item از $B\der A$: \ $B\to a\mid bc$.
\end{itemize}
نتیجه:
\[S\to Aa\mid a\mid bc\mid bb,\qquad A\to a\mid bc\mid bb,\qquad B\to bb\mid a\mid bc.\]
($B$ حالا غیرقابل دسترس و بی‌فایده است و می‌توان آن را حذف کرد.)
\end{example}

\begin{theorem}
اگر $L$ زبانی مستقل از متن باشد و $\eps\notin L$، گرامری برای $L$ وجود دارد که قاعده بی‌فایده، قاعده $\eps$ و قاعده واحد ندارد.
\end{theorem}
\begin{proof}
سه الگوریتم بالا را به این ترتیب اجرا می‌کنیم: (۱) حذف قاعده‌های $\eps$، (۲) حذف قاعده‌های واحد، (۳) حذف قاعده‌های بی‌فایده. این ترتیب درست کار می‌کند چون حذف قاعده‌های واحد قاعده $\eps$ جدیدی نمی‌سازد و حذف قاعده‌های بی‌فایده نه قاعده $\eps$ می‌سازد و نه قاعده واحد (فقط قاعده‌ها را حذف می‌کند).
\end{proof}

\section{فرم نرمال چامسکی}

\begin{definition}[فرم نرمال چامسکی (\lr{CNF})]
گرامر مستقل از متن در \term{فرم نرمال چامسکی}{Chomsky Normal Form} است اگر هر قاعده‌اش به یکی از این دو شکل باشد:
\[A\to BC,\qquad A\to a,\]
که $a$ پایانه است و $A,B,C$ متغیرند و $B$ و $C$ متغیر شروع نیستند. علاوه بر این، قاعده $S\to\eps$ (برای متغیر شروع $S$) هم مجاز است.
\end{definition}

\begin{theorem}\label{thm:cnf}
هر زبان مستقل از متن را یک گرامر در فرم نرمال چامسکی تولید می‌کند.
\end{theorem}

\begin{proofidea}
هر گرامر را با چند گام به \lr{CNF} تبدیل می‌کنیم: متغیر شروع جدید اضافه می‌کنیم، قاعده‌های $\eps$ و واحد را حذف می‌کنیم و قاعده‌های باقی‌مانده را به شکل درست درمی‌آوریم.
\end{proofidea}

\begin{proof}
فرض کنید $G$ گرامری برای زبان باشد.
\begin{enumerate}
\item \textbf{متغیر شروع جدید:} متغیر $S_0$ و قاعده $S_0\to S$ را اضافه کنید. حالا متغیر شروع هرگز در سمت راست قاعده‌ای ظاهر نمی‌شود.
\item \textbf{حذف قاعده‌های $\eps$:} هر قاعده $A\to\eps$ با $A\neq S_0$ را حذف کنید. برای هر ظهور $A$ در سمت راست یک قاعده، نسخه‌ای از آن قاعده را که آن ظهور حذف شده اضافه کنید (برای همه ترکیب‌ها؛ مثلاً از $R\to uAvAw$ نسخه‌های $R\to uvAw$، $R\to uAvw$ و $R\to uvw$ را می‌گیریم). اگر $R\to A$ داشتیم، $R\to\eps$ اضافه می‌شود، مگر اینکه قبلاً $R\to\eps$ را حذف کرده باشیم. این کار را تا وقتی قاعده $\eps$ (جز احتمالاً $S_0\to\eps$) باقی است تکرار کنید.
\item \textbf{حذف قاعده‌های واحد:} هر قاعده $A\to B$ را حذف کنید و برای هر قاعده $B\to u$، قاعده $A\to u$ را اضافه کنید (مگر اینکه $A\to u$ یک قاعده واحد باشد که قبلاً حذف شده). تکرار کنید تا قاعده واحدی نماند.
\item \textbf{شکستن قاعده‌های بلند:} هر قاعده $A\to u_1u_2\cdots u_k$ با $k\ge3$ را با زنجیره‌ای از قاعده‌ها با متغیرهای جدید جایگزین کنید:
\[A\to u_1A_1,\quad A_1\to u_2A_2,\quad\dots,\quad A_{k-2}\to u_{k-1}u_k.\]
\item \textbf{جدا کردن پایانه‌ها:} در هر قاعده با دو نماد سمت راست، هر پایانه $u_i$ را با متغیر جدید $U_i$ جایگزین کنید و قاعده $U_i\to u_i$ را اضافه کنید.
\end{enumerate}
هر گام زبان را حفظ می‌کند (برای گام‌های ۲ و ۳ مثل اثبات‌های بالا، و گام‌های ۴ و ۵ همان قضیه جایگزینی \ref{thm:subst} به صورت برعکس هستند). در پایان هر قاعده شکل $A\to BC$، $A\to a$ یا $S_0\to\eps$ دارد.
\end{proof}

\begin{example}[تبدیل به \lr{CNF}]
گرامر زیر را به \lr{CNF} تبدیل می‌کنیم:
\[S\to ASA\mid aB,\qquad A\to B\mid S,\qquad B\to b\mid\eps.\]
\textbf{۱. متغیر شروع جدید:}
\[S_0\to S,\quad S\to ASA\mid aB,\quad A\to B\mid S,\quad B\to b\mid\eps.\]
\textbf{۲. حذف $B\to\eps$:} از $S\to aB$ قاعده $S\to a$ و از $A\to B$ قاعده $A\to\eps$ به دست می‌آید:
\[S_0\to S,\quad S\to ASA\mid aB\mid a,\quad A\to B\mid S\mid\eps,\quad B\to b.\]
\textbf{حذف $A\to\eps$:} از $S\to ASA$ نسخه‌های $SA$، $AS$ و $S$ ساخته می‌شوند:
\[S_0\to S,\quad S\to ASA\mid aB\mid a\mid SA\mid AS\mid S,\quad A\to B\mid S,\quad B\to b.\]
\textbf{۳. حذف قاعده‌های واحد:} $S\to S$ را حذف می‌کنیم. با حذف $S_0\to S$ همه قاعده‌های $S$ به $S_0$ اضافه می‌شوند. با حذف $A\to B$ قاعده $A\to b$ و با حذف $A\to S$ همه قاعده‌های $S$ به $A$ اضافه می‌شوند:
\[\begin{array}{l}
S_0\to ASA\mid aB\mid a\mid SA\mid AS,\qquad S\to ASA\mid aB\mid a\mid SA\mid AS,\\
A\to b\mid ASA\mid aB\mid a\mid SA\mid AS,\qquad B\to b.
\end{array}\]
\textbf{۴ و ۵. شکل نهایی:} $ASA$ را با $AA_1$ و قاعده جدید $A_1\to SA$ جایگزین می‌کنیم و در $aB$ پایانه $a$ را با متغیر $U$ و قاعده $U\to a$:
\[\begin{array}{l}
S_0\to AA_1\mid UB\mid a\mid SA\mid AS,\qquad S\to AA_1\mid UB\mid a\mid SA\mid AS,\\
A\to b\mid AA_1\mid UB\mid a\mid SA\mid AS,\qquad A_1\to SA,\qquad U\to a,\qquad B\to b.
\end{array}\]
\end{example}

\section{فرم نرمال گرایباخ}

در فرم نرمال گرایباخ محدودیت روی \textbf{جای} پایانه‌ها و متغیرهاست: هر قاعده دقیقاً با یک پایانه شروع می‌شود.

\begin{definition}[فرم نرمال گرایباخ (\lr{GNF})]
گرامر مستقل از متن در \term{فرم نرمال گرایباخ}{Greibach Normal Form} است اگر همه قاعده‌هایش به شکل
\[A\to ax\qquad(a\in T,\ x\in V^*)\]
باشند.
\end{definition}

\begin{theorem}
برای هر گرامر مستقل از متن $G$ با $\eps\notin L(G)$، گرامر هم‌ارزی در فرم نرمال گرایباخ وجود دارد.
\end{theorem}

به‌طور کلی تبدیل یک گرامر به \lr{GNF} و اثبات اینکه این کار همیشه ممکن است ساده نیست و در این درس به آن نمی‌پردازیم. اما در مثال‌های ساده، قضیه جایگزینی \ref{thm:subst} معمولاً کافی است.

\begin{example}
گرامر
\[S\to AB,\qquad A\to aA\mid bB\mid b,\qquad B\to b\]
در \lr{GNF} نیست، چون $S\to AB$ با متغیر شروع می‌شود. با قضیه جایگزینی، $A$ را در $S\to AB$ با سمت راست قاعده‌هایش جایگزین می‌کنیم:
\[S\to aAB\mid bBB\mid bB,\qquad A\to aA\mid bB\mid b,\qquad B\to b.\]
این گرامر در \lr{GNF} است.
\end{example}

\begin{note}
در \lr{GNF} هر گام اشتقاق دقیقاً یک پایانه تولید می‌کند. پس اشتقاق یک رشته به طول $n$ دقیقاً $n$ گام دارد. در فصل بعد از این ایده برای تجزیه سریع استفاده می‌کنیم.
\end{note}

\begin{summary}
\begin{itemize}
\item قضیه جایگزینی: می‌توان یک متغیر در سمت راست را با همه سمت‌راست‌های قاعده‌هایش جایگزین کرد.
\item حذف بی‌فایده‌ها: اول متغیرهای نامولد، بعد متغیرهای غیرقابل دسترس.
\item حذف قاعده‌های $\eps$ با یافتن متغیرهای تهی‌پذیر؛ حذف قاعده‌های واحد با گراف وابستگی.
\item \lr{CNF}: فقط $A\to BC$ و $A\to a$ (و $S\to\eps$). هر \lr{CFL} یک گرامر \lr{CNF} دارد.
\item \lr{GNF}: فقط $A\to ax$ با $x\in V^*$.
\end{itemize}
\end{summary}
